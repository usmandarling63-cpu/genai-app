        response=requests.post(url="https://shaik5195522.app.n8n.cloud/webhook-test/2f8b221c-7161-4ad8-9f84-cc2154fad7b4",json={"input":text,"language":"Hindi"})
        if response.status_code==200:
            st.write(response.json()["text"])
